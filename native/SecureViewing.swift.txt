import WebKit

/// Layer 1 (undocumented Apple behaviour): draw the app inside a secure text-field layer, which iOS
/// excludes from screenshots and screen recordings, so they come out BLANK. If it ever causes a
/// display problem on some iOS version, set this to false. Layer 2 below keeps working either way.
let CS_USE_SECURE_LAYER = false

/// Layer 2 (documented): detect recording / mirroring / screenshots, blank the web view, and tell the
/// page (js/viewer.js -> CSNativeAlert) to end the session and wipe the content.
class SecureViewController: CAPBridgeViewController {
    private var tokens: [NSObjectProtocol] = []
    private var secureApplied = false
    private var secureField: UITextField?

    override func viewDidLoad() {
        super.viewDidLoad()
        let nc = NotificationCenter.default
        tokens.append(nc.addObserver(forName: UIScreen.capturedDidChangeNotification, object: nil, queue: .main) { [weak self] _ in
            self?.captureChanged()
        })
        tokens.append(nc.addObserver(forName: UIApplication.userDidTakeScreenshotNotification, object: nil, queue: .main) { [weak self] _ in
            self?.notifyPage("screenshot")
        })
        // App-switcher snapshot and control-centre pulls: hide while inactive.
        tokens.append(nc.addObserver(forName: UIApplication.willResignActiveNotification, object: nil, queue: .main) { [weak self] _ in
            self?.webView?.isHidden = true
        })
        tokens.append(nc.addObserver(forName: UIApplication.didBecomeActiveNotification, object: nil, queue: .main) { [weak self] _ in
            self?.captureChanged()
        })
    }

    override func viewDidAppear(_ animated: Bool) {
        super.viewDidAppear(animated)
        if CS_USE_SECURE_LAYER && !secureApplied, let window = view.window {
            secureApplied = true
            secureField = window.csMakeSecure()
        }
        captureChanged()
    }

    override func viewDidLayoutSubviews() {
        super.viewDidLayoutSubviews()
        // keep the secure layer exactly the size of the window (rotation, split view), origin 0,0
        if let w = view.window, let f = secureField, f.frame != w.bounds { f.frame = w.bounds }
    }

    private func captureChanged() {
        let captured = UIScreen.main.isCaptured
        webView?.isHidden = captured          // blank at once while recording or mirroring
        webView?.evaluateJavaScript("window.CS_SCREEN_CAPTURED = \(captured ? "true" : "false");", completionHandler: nil)
        if captured { notifyPage("recording") }
    }

    private func notifyPage(_ why: String) {
        webView?.evaluateJavaScript("window.CSNativeAlert && window.CSNativeAlert('\(why)');", completionHandler: nil)
    }
}

extension UIWindow {
    /// The secure field must be exactly the window's size with its origin at 0,0. (A small centred field
    /// shifts everything drawn inside it by half the screen, which pushed the app off to the right/bottom.)
    @discardableResult
    func csMakeSecure() -> UITextField {
        let field = UITextField(frame: bounds)
        field.isSecureTextEntry = true
        field.isUserInteractionEnabled = false
        field.backgroundColor = .clear
        field.borderStyle = .none
        addSubview(field)
        field.layoutIfNeeded()
        layer.superlayer?.addSublayer(field.layer)
        let host: CALayer?
        if #available(iOS 17.0, *) { host = field.layer.sublayers?.last } else { host = field.layer.sublayers?.first }
        host?.frame = field.bounds
        host?.addSublayer(layer)
        return field
    }
}
